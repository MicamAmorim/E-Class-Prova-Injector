\noindent \textbf{MATEMÁTICA}

\begin{enumerate}
\item O valor de $2+2$ é:
\begin{enumerate}
\item $2$
\item $3$
\item $4$
\item $5$
\item $6$
\end{enumerate}

\item A solução de
\[
x+3=7
\]
é:
\begin{enumerate}
\item $x=1$
\item $x=2$
\item $x=3$
\item $x=4$
\item $x=5$
\end{enumerate}
\end{enumerate}
